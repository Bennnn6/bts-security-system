BAGIAN 1: ARSITEKTUR PORTAL PENGGUNA & KONTROL HAK AKSES (RBAC)
Alur otentikasi login Admin, Portal Teknisi, dan proteksi sesi halaman web.
--------------------------------------------------------------------------------
graph TD
    User(["Pengguna Mengakses URL Web"]) --> CheckLogin{"Apakah Memiliki Sesi Login Aktif?"}
    
    CheckLogin -->|Belum Login| LoginPage["Tampilkan Halaman Login (/)"]
    LoginPage --> InputAuth["Pengguna Input Username & Password"]
    
    InputAuth --> CheckRole{"Tipe Akun Pengguna?"}
    
    %% Alur Admin
    CheckRole -->|Akun Admin NOC| AuthAdmin["Validasi ke Database 'admin':<br/>check_password_hash(admin.password_hash, password)"]
    AuthAdmin --> AdminValid{"Password Benar?"}
    AdminValid -->|Salah| ErrLogin["Tolak Login: Flash Message 'Kredensial Tidak Valid'"]
    AdminValid -->|Benar| StartAdminSession["1. login_user(admin)<br/>2. Set session['auth_role'] = 'ADMIN'<br/>3. Timer Sesi: 30 Menit Timeout"]
    StartAdminSession --> AdminDashboard["Redirect ke: /dashboard (NOC Command Center)"]
    
    %% Alur Teknisi
    CheckRole -->|Akun Teknisi Lapangan| AuthTech["Validasi ke Database 'technician':<br/>check_password_hash(tech.password_hash, password)"]
    AuthTech --> TechValid{"Akun Aktif & Password Benar?"}
    TechValid -->|Salah/Nonaktif| ErrLogin
    TechValid -->|Benar| StartTechSession["1. Set session['worker_id'] = tech_id<br/>2. Set session['auth_role'] = 'WORKER'"]
    StartTechSession --> TechDashboard["Redirect ke: /worker/dashboard (Portal Teknisi)"]
    
    %% Proteksi Route (Decorators)
    CheckLogin -->|Sudah Login| RoutingAuth{"Tujuan URL yang Diakses?"}
    RoutingAuth -->|Rute Khusus Admin| GuardAdmin{"@admin_required: Apakah Role == ADMIN?"}
    GuardAdmin -->|Bukan Admin| Denied403["HTTP 403: Akses Ditolak (Khusus Admin NOC)"]
    GuardAdmin -->|Admin Sah| AccessAdminView["Buka Halaman Admin (Sites, Technicians, Logs, Analytics)"]
    
    RoutingAuth -->|Rute Khusus Teknisi| GuardTech{"@technician_required: Apakah Role == WORKER?"}
    GuardTech -->|Bukan Teknisi| Denied403
    GuardTech -->|Teknisi Sah| AccessTechView["Buka Halaman Teknisi (My Work Orders, My Logs, Check-in)"]
    
    RoutingAuth -->|Terminal Kiosk Gerbang| KioskView["Akses Terbuka /kiosk (Kiosk Terminal Gerbang BTS)"]
    
    %% Logout
    AccessAdminView --> UserLogout["Pengguna Klik Tombol Logout (/logout)"]
    AccessTechView --> UserLogout
    UserLogout --> DestroySession["Hapus Sesi Flask-Login & Cookie -> Redirect ke Halaman Login"]

BAGIAN 5: ALUR DETAIL AUTENTIKASI AKSES GERBANG & IOT HARDWARE RELAY
Logika mendalam verifikasi multi-faktor (Face, Liveness, Geofence, Work Order, dan Webhook).
--------------------------------------------------------------------------------
graph TD
    Start(["Teknisi Berada di Depan Kiosk Menara BTS"]) --> RequestAccess["Teknisi Klik 'Buka Gerbang' di Web Kiosk"]
    RequestAccess --> CaptureData["Kiosk Mengambil:<br/>- Snapshot Kamera Wajah (Base64)<br/>- Geolocation GPS HP/Kiosk (Lat, Lon)<br/>- ID Menara (site_id)<br/>- Tipe Akses ('MASUK' / 'KELUAR')"]
    
    CaptureData --> PostVerify["Kirim HTTP POST /api/verify-face"]
    
    %% Validasi Tahap 1: Site & GPS
    PostVerify --> V1_Site{"1. Validasi Menara:<br/>BtsSite Terdaftar & Status 'Normal (Up)'?"}
    V1_Site -->|Tidak| Denied_Site["HTTP 404/403: SITE_NOT_FOUND / SITE_DOWN"]
    
    V1_Site -->|Ya| V2_Geo{"2. Validasi Geofencing GPS:<br/>Jarak Haversine <= 150 Meter?"}
    V2_Geo -->|Jarak > 150m| Denied_Geo["HTTP 403: OUT_OF_GEOFENCE<br/>Kirim Telegram Alert 'Pelanggaran Geofence'"]
    
    %% Validasi Tahap 2: Anti-Spoofing Liveness
    V2_Geo -->|Radius Aman| V3_FaceLoc{"3. Deteksi Wajah HOG:<br/>Tepat 1 Wajah Terlihat?"}
    V3_FaceLoc -->|0 Wajah / >1 Wajah| Denied_FaceLoc["HTTP 400: FACE_NOT_FOUND / MULTIPLE_FACES"]
    
    V3_FaceLoc -->|1 Wajah| V4_Liveness{"4. Uji Biometrik Liveness Pasif:<br/>- Laplacian Variance (Tekstur)<br/>- Specular Glare (Pantulan Layar HP)<br/>- YCrCb Chromaticity (Rona Kulit)<br/>- Eye Aspect Ratio (Landmark Mata)<br/>Skor >= 0.55 & No Glare Penalty?"}
    V4_Liveness -->|Gagal / Foto HP| Denied_Spoof["HTTP 403: SPOOF_ATTACK_DETECTED<br/>Kirim Telegram Alert Anti-Spoofing + Foto"]
    
    %% Validasi Tahap 3: Pencocokan Wajah
    V4_Liveness -->|Lolos (Manusia Asli)| V5_Vector["5. Ekstraksi Vektor Wajah 128-D"]
    V5_Vector --> V6_Matching{"6. Komparasi Euclidean Distance vs Teknisi Aktif:<br/>best_distance <= Threshold (0.45/0.52)?"}
    V6_Matching -->|Tidak Cocok| Denied_Unknown["HTTP 403: FACE_UNKNOWN<br/>Kirim Telegram Alert 'Wajah Asing'"]
    
    %% Validasi Tahap 4: Aturan & Izin Kerja
    V6_Matching -->|Wajah Cocok| V7_Passback{"7. Cek Anti-Passback:<br/>Status Masuk/Keluar Wajar?"}
    V7_Passback -->|Dobel Masuk/Keluar| Denied_Passback["HTTP 409: INVALID_ACCESS_STATE"]
    
    V7_Passback -->|Status Wajar| V8_Cooldown{"8. Cek Cooldown Gerbang:<br/>Gate Baru Terbuka < 3 Detik Lalu?"}
    V8_Cooldown -->|Masih Cooldown| Denied_Cooldown["HTTP 429: GATE_COOLDOWN"]
    
    V8_Cooldown -->|Siap| V9_WO{"9. Pengecekan Work Order:<br/>Ada Tiket APPROVED Hari Ini?"}
    V9_WO -->|Tidak Ada Izin| Denied_WO["HTTP 403: NO_ACTIVE_WORK_ORDER"]
    
    %% Eksekusi Sukses
    V9_WO -->|Tiket Valid / Keluar| Action_Grant["AKSES DIIJINKAN (GRANTED)!"]
    
    Action_Grant --> OpenGateFunc["Panggil open_gate(device_id, site_id)"]
    OpenGateFunc --> TriggerRelay["Thread Daemon Asinkron: Kirim Webhook HMAC-SHA256<br/>ESP32 Mengaktifkan Relay Solenoid -> Kunci Terbuka"]
    TriggerRelay --> UpdateWODB["Otomatis Ubah Status Work Order -> 'IN_PROGRESS'"]
    UpdateWODB --> LogAccessDB["Simpan AccessLog (Result: GRANTED, gate_status: 'OPEN')"]
    LogAccessDB --> SendTelegramPass["Kirim Notifikasi Telegram Bot NOC (Async)"]
    SendTelegramPass --> TimerAutoClose["Timer Tutup Otomatis (10-30 Detik)"]
    TimerAutoClose --> AutoClosed["Kunci Solenoid Gerbang Menutup Rapat Kembali"]
    
    %% Blok Penolakan Bersama
    Denied_Site --> RecordDenied["Simpan AccessLog (Result: DENIED) & Catat ActivityLog"]
    Denied_Geo --> RecordDenied
    Denied_FaceLoc --> RecordDenied
    Denied_Spoof --> RecordDenied
    Denied_Unknown --> RecordDenied
    Denied_Passback --> RecordDenied
    Denied_Cooldown --> RecordDenied
    Denied_WO --> RecordDenied

BAGIAN 6: ALUR CADANGAN OTP, KEYPAD & PROTOKOL SILENT DURESS PIN
Otentikasi alternatif jika kamera bermasalah dan perlindungan darurat penyanderaan.
--------------------------------------------------------------------------------
graph TD
    Start(["Teknisi Pilih Opsi 'Gunakan Kode OTP' di Kiosk"]) --> FormOTP["Tampilkan Form Keypad:<br/>Input: ID Teknisi & 6 Digit PIN OTP"]
    
    FormOTP --> PostOTP["Kirim HTTP POST /api/verify-otp"]
    
    PostOTP --> CheckLockout{"Cek Riwayat Gagal Teknisi Ini:<br/>Apakah Gagal >= 3x dalam 5 Menit Terakhir?"}
    CheckLockout -->|Ya (Terkunci)| LockResponse["HTTP 429: OTP_LOCKED<br/>Kembalikan Pesan: 'Akun Terkunci Selama 5 Menit'"]
    
    CheckLockout -->|Tidak Terkunci| CheckDuress{"Apakah PIN == DURESS_PIN?<br/>(Default: '999999')"}
    
    %% PROTOKOL SILENT DURESS (SANDERA)
    CheckDuress -->|YA (KONDISI DISANDERA / DITODONG)| DuressProtocol["🚨 PROTOKOL PENYELAMATAN SANDERA AKTIF:"]
    DuressProtocol --> D1["1. Layar Kiosk Tetap Tampil 'Sukses' & Gerbang Terbuka Normal<br/>(Agar Pelaku Tidak Menyakiti Teknisi)"]
    DuressProtocol --> D2["2. Simpan AccessLog: reason='SILENT_DURESS_ALARM', level='CRITICAL'"]
    DuressProtocol --> D3["3. Kamera Mengambil Foto Wajah Pelaku Diam-Diam"]
    DuressProtocol --> D4["4. Kirim Telegram Alert Darurat ke Tim Patroli & Kepolisian:<br/>⚠️ PERINGATAN DARURAT: TEKNISI DISANDERA DI SITE BTS!"]
    D1 --> GateOpenDuress(["Gerbang Terbuka (Teknisi Selamat)"])
    
    %% ALUR OTP STANDAR
    CheckDuress -->|TIDAK (Otentikasi Biasa)| VerifyTOTPAlgorithm{"Verifikasi Algoritma RFC 6238 TOTP:<br/>verify_totp(technician.totp_secret, otp, valid_window=1)"}
    
    VerifyTOTPAlgorithm -->|Kode Salah / Expired| HandleFail["Catat Percobaan Gagal: Failures + 1"]
    HandleFail --> CheckMaxFail{"Total Gagal >= 3 Kali?"}
    CheckMaxFail -->|Ya| LockAccount["Kunci Akses Teknisi 5 Menit<br/>Catat ActivityLog Level: HIGH"]
    CheckMaxFail -->|Belum| ReturnWrongMsg["HTTP 403: OTP_INVALID<br/>Beri Tahu Sisa Kesempatan (Contoh: Sisa 2 Kali)"]
    
    VerifyTOTPAlgorithm -->|Kode Benar| ResetFail["Reset Counter Gagal -> 0"]
    ResetFail --> CheckWOStep["Cek Surat Perintah Kerja (Work Order)"]
    CheckWOStep --> OpenGateSuccess["Buka Gerbang Fisik & Catat Log GRANTED (NORMAL)"]
