MODUL 1: VALIDASI AWAL, PAYLOAD & GPS GEOFENCING (HAVERSINE)
Memeriksa format data, status operasional menara, dan radius jarak GPS teknisi.
--------------------------------------------------------------------------------
graph TD
    Start(["Teknisi Mengajukan Akses di Kiosk/App"]) --> RecvReq["Menerima Request HTTP POST /api/verify-face"]
    RecvReq --> CheckJSON{"Format Content-Type JSON & Body Valid?"}
    
    CheckJSON -->|Tidak Valid| ErrJSON["HTTP 415/400: JSON_INVALID<br/>Log: Akses Ditolak"]
    CheckJSON -->|Valid| ParseData["Ekstraksi Data:<br/>- image (Base64)<br/>- site_id<br/>- access_type (MASUK/KELUAR)<br/>- latitude & longitude"]
    
    ParseData --> QuerySite["Query Database:<br/>BtsSite.query.filter_by(site_id)"]
    QuerySite --> SiteFound{"Menara Terdaftar di Database?"}
    
    SiteFound -->|Tidak| ErrSite404["HTTP 404: SITE_NOT_FOUND<br/>Catat ActivityLog WARNING"]
    SiteFound -->|Ya| CheckSiteUp{"Status Operasional Menara?<br/>(status == 'Normal (Up)')"}
    
    CheckSiteUp -->|Maintenance / Down| ErrSiteDown["HTTP 403: SITE_DOWN<br/>Akses Gerbang Dinonaktifkan"]
    CheckSiteUp -->|Normal (Up)| CheckGeoConfig{"GEOFENCE_ENABLED == True?<br/>(Koordinat GPS Dikirim?)"}
    
    CheckGeoConfig -->|Tidak Ada GPS / Nonaktif| BypassGeo["Bypass Geofencing -> Lanjut Validasi Foto"]
    CheckGeoConfig -->|Aktif| CalcHaversine["Hitung Jarak Menggunakan Formula Haversine:<br/>d = 2R * arcsin(sqrt(sin^2(dlat/2) + cos*cos*sin^2(dlon/2)))"]
    
    CalcHaversine --> CheckRadius{"Jarak <= GEOFENCE_RADIUS_METERS?<br/>(Default: Maksimal 150 Meter)"}
    CheckRadius -->|Jarak > 150m| ErrGeo["HTTP 403: OUT_OF_GEOFENCE<br/>1. Simpan AccessLog (Result: DENIED, Level: HIGH)<br/>2. Catat ActivityLog: GEOFENCE_VIOLATION<br/>3. Kirim Telegram Alert 'Pelanggaran Geofence'"]
    
    CheckRadius -->|Dalam Radius (<= 150m)| CheckImg["Validasi Base64 Gambar:<br/>validate_evidence(image) <= 2MB"]
    BypassGeo --> CheckImg
    
    CheckImg -->|Gambar Kosong / Rusak| ErrImg["HTTP 400: FACE_IMAGE_MISSING<br/>Log DENIED (Level: WARNING)"]
    CheckImg -->|Gambar Valid| NextModul2(["Lolos Tahap 1: Lanjut ke Modul 2 (Anti-Spoofing)"])

MODUL 2: ENGINE ANTI-SPOOFING & LIVENESS DETECTION (COMPUTER VISION)
Mendeteksi apakah wajah di depan kamera manusia asli atau foto/layar HP.
--------------------------------------------------------------------------------
graph TD
    Start(["Menerima Gambar Wajah (RGB)"]) --> RunHOG["Jalankan Detektor Wajah:<br/>face_recognition.face_locations(model='hog')"]
    RunHOG --> CountFaces{"Berapa Jumlah Wajah Terdeteksi?"}
    
    CountFaces -->|0 Wajah| ErrNoFace["HTTP 400: FACE_NOT_FOUND<br/>Wajah Tidak Pas di Tengah Kamera"]
    CountFaces -->|> 1 Wajah| ErrMultiFace["HTTP 400: MULTIPLE_FACES<br/>Terdeteksi Lebih dari 1 Orang"]
    
    CountFaces -->|Tepat 1 Wajah| CropROI["Crop Bounding Box ROI Wajah:<br/>Ekstraksi Area Wajah untuk Analisis"]
    
    CropROI --> TestLaplacian["1. Uji Ketajaman Tekstur (Laplacian Var):<br/>- var < 30: Buram / Kertas Jauh (Score Rendah)<br/>- var > 2800: Grid LCD / Moiré Pattern (Score Rendah)<br/>- 40 <= var <= 1800: Tekstur Alami Kulit Manusia"]
    
    CropROI --> TestGlare["2. Uji Pantulan Layar Kaca (Specular Glare):<br/>- Konversi ke HSV: (V > 248 & S < 35)<br/>- Hitung Rasio Area Pantulan / Luas Wajah<br/>- Jika glare_ratio > 0.05 -> Pinalti Glare Ekstrem"]
    
    CropROI --> TestSkin["3. Uji Rona Warna Kulit (YCrCb Chromaticity):<br/>- Nilai Cr: 130 s/d 175, Nilai Cb: 75 s/d 130<br/>- Hitung skin_ratio: Rasio Pigmen Alami Kulit"]
    
    CropROI --> TestLandmarks["4. Uji Biometrik Landmark Mata (EAR):<br/>- Ekstraksi 6 Koordinat per Mata<br/>- EAR = (||p2-p6|| + ||p3-p5||) / (2 * ||p1-p4||)<br/>- Rentang Manusia Normal: 0.16 s/d 0.42"]
    
    TestLaplacian --> CalcComposite["Hitung Skor Komposit Liveness:<br/>Skor = (40% Tekstur) + (35% Rona Kulit) + (25% EAR) - Pinalti Glare"]
    TestGlare --> CalcComposite
    TestSkin --> CalcComposite
    TestLandmarks --> CalcComposite
    
    CalcComposite --> EvalLiveness{"Skor >= LIVENESS_MIN_SCORE (0.55)<br/>& Glare Penalty <= 0.20?"}
    
    EvalLiveness -->|Gagal (Terdeteksi Foto/Layar)| DeniedSpoof["HTTP 403: SPOOF_ATTACK_DETECTED<br/>1. Simpan AccessLog (Result: DENIED, Level: HIGH)<br/>2. Catat ActivityLog: Percobaan Pemalsuan Wajah<br/>3. Kirim Telegram Alert Anti-Spoofing + Bukti Foto"]
    
    EvalLiveness -->|Lolos (Manusia Hidup Asli)| NextModul3(["Lolos Tahap 2: Lanjut ke Modul 3 (Pencocokan Wajah)"])

    ValidVector -->|Tidak| ErrEnc["HTTP 500: FACE_PROCESSING_ERROR<br/>Gagal Mengekstraksi Fitur Wajah"]
    
    ValidVector -->|Ya| QueryTech["Ambil Seluruh Teknisi Status 'Active':<br/>Technician.query.filter_by(status='Active')"]
    
    QueryTech --> AnyTech{"Apakah Ada Teknisi Aktif Terdaftar?"}
    AnyTech -->|Tidak Ada| ErrNoTech["HTTP 503: NO_ACTIVE_TECHNICIAN<br/>Belum Ada Akun Teknisi Aktif"]
    
    AnyTech -->|Ada| LoopMatch["Iterasi Pencocokan Wajah Teknisi:<br/>Bandingkan dengan Setiap face_encoding Database"]
    
    LoopMatch --> CalcDist["Hitung Euclidean Distance:<br/>distance = face_distance([known_enc], current_enc)[0]"]
    CalcDist --> FindMin["Cari Teknisi dengan Jarak Terkecil (best_distance)"]
    
    FindMin --> CheckThreshold{"best_distance <= FACE_MATCH_THRESHOLD?<br/>(Default: 0.45 s/d 0.52)"}
    
    CheckThreshold -->|Jarak Lebih Besar / Tidak Cocok| DeniedUnknown["HTTP 403: FACE_UNKNOWN<br/>1. Simpan AccessLog (Result: DENIED, Level: HIGH)<br/>2. Catat ActivityLog: Wajah Tidak Dikenali<br/>3. Kirim Telegram Security Alert + Foto Wajah Asing"]
    
    CheckThreshold -->|Jarak Masuk Ambang Batas| Matched(["Wajah Teridentifikasi!<br/>Teknisi: best_technician<br/>Lanjut ke Modul 4 (Aturan Akses)"])
