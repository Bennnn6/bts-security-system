flowchart TD

    %% --------------------------------------------------------
    %% SUBGRAPH 1: INISIASI & LOKASI
    %% --------------------------------------------------------
    subgraph S1 ["1. Inisiasi Akses & Validasi Lokasi (GPS)"]
        Start(["Teknisi Tiba di Menara BTS"]) --> PilihMetode{"Pilih Metode Akses di Kiosk/App"}
        PilihMetode -->|Scan Wajah & Lokasi| InFace["Kirim Foto Kamera + Koordinat GPS"]
        PilihMetode -->|Keypad Kode Darurat/OTP| InOTP["Kirim ID Teknisi + 6 Digit Kode PIN/OTP"]

        InFace --> CekSite{"Validasi Menara BTS:<br/>Site Terdaftar & Aktif?"}
        CekSite -->|Tidak| DeniedSite["Akses Ditolak:<br/>Menara Tidak Terdaftar/Down"]
        CekSite -->|Ya| CekGeo{"Validasi GPS Geofencing:<br/>Jarak <= 150m dari Tower?"}
        CekGeo -->|Jarak > 150m| DeniedGeo["Akses Ditolak:<br/>Teknisi di Luar Radius Menara"]
        CekGeo -->|Dalam Radius| InLiveness["Mulai Uji Keaslian Wajah"]
    end

    %% --------------------------------------------------------
    %% SUBGRAPH 2: ANTI-SPOOFING & LIVENESS
    %% --------------------------------------------------------
    subgraph S2 ["2. Evaluasi Anti-Spoofing & Keaslian Wajah (Liveness)"]
        InLiveness --> DetFace{"Deteksi Wajah (HOG Model)"}
        DetFace -->|0 atau >1 Wajah| DeniedFaceCount["Akses Ditolak:<br/>Wajah Tidak Pas di Kamera"]
        DetFace -->|Tepat 1 Wajah| FilterLive["Uji 4 Parameter Biometrik:<br/>1. Laplacian Tekstur Alami<br/>2. Deteksi Pantulan Layar Kaca HP<br/>3. Rasio Rona Warna Kulit YCrCb<br/>4. Landmark Mata (EAR Ratio)"]
        FilterLive --> SkorLive{"Skor Liveness >= 0.55 &<br/>Bebas Pantulan Layar HP?"}
        SkorLive -->|Gagal / Pakai Layar HP| DeniedSpoof["Akses Ditolak:<br/>Terindikasi Pemalsuan Foto/Layar!"]
        SkorLive -->|Lolos (Manusia Asli)| Extract128["Ekstraksi Vektor Wajah 128-Dimensi"]
    end

    %% --------------------------------------------------------
    %% SUBGRAPH 3: PENCOCOKAN WAJAH & ANTI-PASSBACK
    %% --------------------------------------------------------
    subgraph S3 ["3. Pencocokan Identitas & Status Akses"]
        Extract128 --> MatchFace{"Bandingkan Jarak Euclidean<br/>dengan Database Teknisi Aktif"}
        MatchFace -->|Jarak > Threshold (0.45/0.52)| DeniedUnknown["Akses Ditolak:<br/>Wajah Tidak Terdaftar"]
        MatchFace -->|Cocok / Teridentifikasi| CekPassback{"Cek Anti-Passback:<br/>Status Masuk/Keluar Valid?"}
        CekPassback -->|Status Dobel| DeniedPassback["Akses Ditolak:<br/>Status Akses Tidak Wajar"]
        CekPassback -->|Valid| CekCooldown{"Apakah Gerbang Sedang Cooldown?"}
        CekCooldown -->|Ya (Masih Terbuka)| DeniedCool["Akses Ditunda:<br/>Gerbang Masih Cooldown"]
        CekCooldown -->|Tidak (Siap)| CekWO["Lanjut ke Pengecekan Izin Kerja"]
    end

    %% --------------------------------------------------------
    %% SUBGRAPH 4: JALUR CADANGAN OTP & SILENT DURESS PIN
    %% --------------------------------------------------------
    subgraph S4 ["4. Jalur Cadangan OTP & Silent Duress PIN (Darurat)"]
        InOTP --> CekLock{"Apakah Gagal >= 3x<br/>dalam 5 Menit?"}
        CekLock -->|Ya| DeniedLocked["Akses Terkunci 5 Menit"]
        CekLock -->|Tidak| CekDuress{"Apakah Kode == PIN Sandera?<br/>(DURESS_PIN: 999999)"}
        
        %% Cabang Silent Duress
        CekDuress -->|YA (KONDISI DITODONG/DISANDERA)| DuressTrigger["1. Gerbang Terbuka Normal di Layar Kiosk<br/>2. Catat Log Darurat SILENT_DURESS_ALARM<br/>3. Kirim Telegram Alert CRITICAL ke Polisi/NOC"]
        DuressTrigger --> ExecOpen
        
        %% Cabang OTP Standar
        CekDuress -->|TIDAK (Normal)| VerifyTOTP{"Verifikasi Kode TOTP RFC 6238"}
        VerifyTOTP -->|Kode Salah| IncFail["Tambah Hitungan Gagal (+1)"]
        IncFail --> DeniedOTP["Kode OTP Salah"]
        VerifyTOTP -->|Kode Benar| CekWO
    end

    %% --------------------------------------------------------
    %% SUBGRAPH 5: SURAT TUGAS & WORK ORDER
    %% --------------------------------------------------------
    subgraph S5 ["5. Validasi Surat Perintah Kerja (Work Order)"]
        CekWO --> CekWOTiket{"Apakah Teknisi Memiliki Tiket<br/>APPROVED di Site Ini Hari Ini?"}
        CekWOTiket -->|Tidak Ada Jadwal / Izin| DeniedWO["Akses Ditolak:<br/>Tidak Memiliki Izin Kerja Resmi"]
        CekWOTiket -->|Tiket Valid / Mode Bebas| GrantAccess["AKSES DIIJINKAN (GRANTED)"]
    end

    %% --------------------------------------------------------
    %% SUBGRAPH 6: EKSEKUSI GERBANG & IOT RELAY
    %% --------------------------------------------------------
    subgraph S6 ["6. Eksekusi Pembukaan Gerbang & IoT Relay"]
        GrantAccess --> ExecOpen["Kirim Sinyal Buka Gerbang:<br/>Trigger Webhook Relay IoT (HMAC-SHA256)"]
        ExecOpen --> UpdateWO["Update Status Work Order -> IN_PROGRESS"]
        UpdateWO --> LogGrant["Simpan AccessLog (Result: GRANTED)"]
        LogGrant --> NotifSuccess["Kirim Laporan Telegram NOC (Async)"]
        NotifSuccess --> AutoClose["Timer Otomatis Gerbang Menutup (10-30 Detik)"]
        AutoClose --> EndSuccess(["Akses Selesai (Gerbang Terkunci Kembali)"])
    end

    %% --------------------------------------------------------
    %% SUBGRAPH 7: PENANGANAN AKSES DITOLAK & KEAMANAN
    %% --------------------------------------------------------
    subgraph S7 ["7. Penanganan Akses Ditolak & Eskalasi Insiden"]
        DeniedSite --> LogDenied["Simpan AccessLog (DENIED) & Catat ActivityLog"]
        DeniedGeo --> LogDenied
        DeniedFaceCount --> LogDenied
        DeniedSpoof --> LogDenied
        DeniedUnknown --> LogDenied
        DeniedPassback --> LogDenied
        DeniedCool --> LogDenied
        DeniedLocked --> LogDenied
        DeniedOTP --> LogDenied
        DeniedWO --> LogDenied

        LogDenied --> CekSeverity{"Tingkat Keparahan Insiden >= HIGH?"}
        CekSeverity -->|Ya (Serangan/Pelanggaran)| TelegramAlert["Kirim Telegram Security Alert + Foto Bukti Kamera"]
        CekSeverity -->|Tidak (Peringatan Biasa)| EndDenied(["Gerbang Tetap Terkunci"])
        TelegramAlert --> EndDenied
    end

    %% --------------------------------------------------------
    %% SUBGRAPH 8: SENSOR FISIK PINTU GERBANG
    %% --------------------------------------------------------
    subgraph S8 ["8. Pengawasan Sensor Pintu Fisik & Pembobolan"]
        ReedSensor["Sensor Magnet Pintu (Reed Switch)"] --> EventSensor{"Status Sensor Pintu?"}
        EventSensor -->|Pintu Dibuka Normal| SensorOpen["Status: Pintu Terbuka"]
        EventSensor -->|Pintu Ditutup Rapat| SensorClose["Status: Pintu Tertutup (NORMAL)"]
        EventSensor -->|Pintu Terbuka Terlalu Lama| SensorAjar["Peringatan: DOOR_AJAR (WARNING)"]
        EventSensor -->|Pintu Dibuka Paksa Tanpa Izin| SensorBreach["🚨 PEMBOBOLAN GERBANG (DOOR_FORCED)"]
        
        SensorBreach --> ActionBreach["1. Bunyikan Sirine Lokal via Relay<br/>2. Catat Log Insiden CRITICAL<br/>3. Kirim Telegram Alert Darurat + Snapshot Kamera"]
        ActionBreach --> LiveMap["Perbarui Tampilan Peta GIS Live NOC"]
        SensorAjar --> LiveMap
        SensorOpen --> LiveMap
        SensorClose --> LiveMap
    end
